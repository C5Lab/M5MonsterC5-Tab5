cmake_minimum_required(VERSION 3.16)
project(tab5_ui_render C CXX)
set(CMAKE_C_STANDARD 11)
set(LV_BUILD_CONF_PATH /work/lv_conf.h CACHE PATH "" FORCE)
set(CONFIG_LV_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_BUILD_DEMOS OFF CACHE BOOL "" FORCE)
add_subdirectory(/repo/managed_components/lvgl__lvgl /tmp/tab5-lvgl)
file(GLOB COMPONENT_INCLUDES /repo/components/*/include)
file(GLOB SUBGHZ_SOURCES /repo/main/screens/subghz*.c)
add_executable(render generated.c /repo/main/lv_font_lato_regular_ttf.c /repo/main/screens/cgw_parser.c
    /repo/main/screens/psram_dynarr.c ${SUBGHZ_SOURCES})
target_include_directories(render PRIVATE /work /work/shims /repo/main/screens ${COMPONENT_INCLUDES})
target_compile_options(render PRIVATE -w -g -ffunction-sections -fdata-sections -fsanitize=address)
target_link_options(render PRIVATE -Wl,--gc-sections -fsanitize=address)
target_link_libraries(render PRIVATE lvgl m)
